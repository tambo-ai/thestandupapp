   ```typescript
   const fullDb = getFullDb();
   const invite = await fullDb
     .selectFrom('invite_links')
     .innerJoin('teams', 'teams.id', 'invite_links.team_id')
     .where('invite_links.token', '=', token)
     .where('invite_links.revoked_at', 'is', null)
     .select(['invite_links.id', 'invite_links.team_id', 'invite_links.max_uses',
              'invite_links.use_count', 'invite_links.expires_at',
              'teams.name as team_name', 'teams.workos_organization_id'])
     .executeTakeFirst();
   ```
