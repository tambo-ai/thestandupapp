**src/app/api/connections/status/route.ts** — Create new file. GET endpoint that returns connection status for the authenticated user.

```typescript
import { getWorkOS, withAuth } from '@workos-inc/authkit-nextjs';
import { NextResponse } from 'next/server';

export async function GET() {
  const { user } = await withAuth({ ensureSignedIn: true });
  const workos = getWorkOS();

  const [github, linear] = await Promise.all([
    workos.pipes.getAccessToken({ provider: 'github', userId: user.id })
      .catch(() => ({ active: false as const, error: 'not_installed' as const })),
    workos.pipes.getAccessToken({ provider: 'linear', userId: user.id })
      .catch(() => ({ active: false as const, error: 'not_installed' as const })),
  ]);

  return NextResponse.json({
    github: github.active ? 'connected' : (github as { error: string }).error ?? 'not_installed',
    linear: linear.active ? 'connected' : (linear as { error: string }).error ?? 'not_installed',
  });
}
```
